12void uart_set_baud_rate(UART_Handle_t* pUART_Handle){
13 uint32_t fclk = RCC_GetUSARTClock(pUART_Handle->pUARTx);
14 if(fclk == 0){
15 return;
16 }
17 uint32_t temp = fclk*125/(1*(2-pUART_Handle->UART_Config.UART_CONFIG_OVER_SAMPLING)*pUART_Handle->UART_Config.UART_CONFIG_BAUD_RATE);
18 uint32_t matissa = temp/1000;
19 uint32_t fraction= (temp-(matissa*1000))*8*(2-pUART_Handle->UART_Config.UART_CONFIG_OVER_SAMPLING)+500;
20 fraction = fraction / 1000;
21 pUART_Handle->pUARTx->USART_BRR &= ~(0xFFFF);
22 pUART_Handle->pUARTx->USART_BRR |= (matissa<<4);
23 pUART_Handle->pUARTx->USART_BRR |= (fraction);
24}

26void uart_init(UART_Handle_t* pUART_Handle)
27{
28 //0. Basic Check first
29 if((pUART_Handle == NULL) || (pUART_Handle->pUARTx == NULL)){
30 return;
31 }
32 //1.Enable the Peripheral Clock before writing to any of the Configuration Register
33 if(pUART_Handle->pUARTx == UART1){
34 USART1_PCLK_EN();
35 }else if(pUART_Handle->pUARTx == UART2){
36 USART2_PCLK_EN();
37 }else if(pUART_Handle->pUARTx == UART3){
38 USART3_PCLK_EN();
39 }else if(pUART_Handle->pUARTx == UART4){
40 UART4_PCLK_EN();
41 }else if(pUART_Handle->pUARTx == UART5){
42 UART5_PCLK_EN();
43 }else if(pUART_Handle->pUARTx == UART6){
44 USART6_PCLK_EN();
45 }
46
47 //2. Set Over Sampling (@note This step must come before setting the baud rate)
48 pUART_Handle->pUARTx->USART_CR1 &= ~(1<<USART_CR1_OVER8);
49 pUART_Handle->pUARTx->USART_CR1 |= (pUART_Handle->UART_Config.UART_CONFIG_OVER_SAMPLING<<USART_CR1_OVER8);
50
51 //3. Set the mode
52 if(pUART_Handle->UART_Config.UART_CONFIG_MODE == UART_CONFIG_MODE_RX){
53 pUART_Handle->pUARTx->USART_CR1 &= ~(1U << USART_CR1_TE);
54 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_RE);
55 }else if (pUART_Handle->UART_Config.UART_CONFIG_MODE == UART_CONFIG_MODE_TX){
56 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_TE);
57 pUART_Handle->pUARTx->USART_CR1 &= ~(1U << USART_CR1_RE);
58 }else if(pUART_Handle->UART_Config.UART_CONFIG_MODE == UART_CONFIG_MODE_TX_RX){
59 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_TE);
60 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_RE);
61 }
62
63 //4. Set Baud Rate
64 uart_set_baud_rate(pUART_Handle);
65
66 //5. Set the parity
67 if(pUART_Handle->UART_Config.UART_CONFIG_PARITY == UART_CONFIG_PARITY_NONE){
68 pUART_Handle->pUARTx->USART_CR1 &= ~(1U << USART_CR1_PCE);
69 }else if(pUART_Handle->UART_Config.UART_CONFIG_PARITY == UART_CONFIG_PARITY_EVEN ){
70 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_PCE);
71 pUART_Handle->pUARTx->USART_CR1 &= ~(1 << USART_CR1_PS);
72 }else if(pUART_Handle->UART_Config.UART_CONFIG_PARITY == UART_CONFIG_PARITY_ODD){
73 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_PCE);
74 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_PS);
75 }
76
77 //6. Set No of stop Bits
78 pUART_Handle->pUARTx->USART_CR2 &= ~(0x3 << USART_CR2_STOP);
79 pUART_Handle->pUARTx->USART_CR2 |= (pUART_Handle->UART_Config.UART_CONFIG_NO_OF_STOP_BITS << USART_CR2_STOP);
80
81 //7. Set Word Length
82 pUART_Handle->pUARTx->USART_CR1 &= ~(0x1<< USART_CR1_M);
83 pUART_Handle->pUARTx->USART_CR1 |= ((pUART_Handle->UART_Config.UART_CONFIG_WORD_LENGTH)<< USART_CR1_M);
84
85 //8. Set Hardware Flow
86 if(pUART_Handle->UART_Config.UART_CONFIG_HW_FLOW_CONTROL == UART_CONFIG_HW_FLOW_CONTROL_NONE){
87 pUART_Handle->pUARTx->USART_CR3 &= ~(1U<< USART_CR3_CTSE);
88 pUART_Handle->pUARTx->USART_CR3 &= ~(1U<< USART_CR3_RTSE);
89 }else if(pUART_Handle->UART_Config.UART_CONFIG_HW_FLOW_CONTROL == UART_CONFIG_HW_FLOW_CONTROL_CTS){
90 pUART_Handle->pUARTx->USART_CR3 |= (1U<< USART_CR3_CTSE);
91 pUART_Handle->pUARTx->USART_CR3 &= ~(1U<< USART_CR3_RTSE);
92 }else if(pUART_Handle->UART_Config.UART_CONFIG_HW_FLOW_CONTROL == UART_CONFIG_HW_FLOW_CONTROL_RTS){
93 pUART_Handle->pUARTx->USART_CR3 &= ~(1U<< USART_CR3_CTSE);
94 pUART_Handle->pUARTx->USART_CR3 |= (1U<< USART_CR3_RTSE);
95 }else if(pUART_Handle->UART_Config.UART_CONFIG_HW_FLOW_CONTROL == UART_CONFIG_HW_FLOW_CONTROL_CTS_RTS){
96 pUART_Handle->pUARTx->USART_CR3 |= (1U<< USART_CR3_CTSE);
97 pUART_Handle->pUARTx->USART_CR3 |= (1U<< USART_CR3_RTSE);
98 }

102 if(pUARTx == NULL){

107 }else{
108 pUARTx->USART_CR1 &= ~(1U << USART_CR1_UE);
109 while (pUARTx->USART_CR1 & (1U << USART_CR1_UE)) {
110 /* wait until UART is disabled */
111 }

144void UART_SendData_Blocking(UART_Handle_t *pUARTHandle, uint8_t *TXBuffer, uint32_t Len) {
145 if(pUARTHandle == NULL || pUARTHandle->pUARTx == NULL || TXBuffer == NULL || Len == 0){
146 return;
147 }
148 uint32_t idx = 0;
149 while (Len > 0) {
150 if (UART_Get_Status_Flag(pUARTHandle->pUARTx, UART_STATUS_FLAG_TXE)) {
151 pUARTHandle->pUARTx->USART_SR &= ~(1U << USART_SR_TC);
152 uint8_t temp = TXBuffer[idx++];
153 pUARTHandle->pUARTx->USART_DR = temp;
154 Len--;
155 }
156 }
157 while (!UART_Get_Status_Flag(pUARTHandle->pUARTx, UART_STATUS_FLAG_TC)) {
158
159 }
160}
