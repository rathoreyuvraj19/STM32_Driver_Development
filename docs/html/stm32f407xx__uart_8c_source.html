12void UART_Echo(UART_Handle_t *pUARTHandle, char ch){
13 if (ch == '\b' || ch == 0x7F) {
14 UART_Send_String(pUARTHandle, "\b \b", 3);
15 }
16 else if (ch == '\r') {
17 UART_Send_String(pUARTHandle, "\r\n", 2);
18 }
19 else {
20 UART_SendData_Blocking(pUARTHandle, (uint8_t*)&ch, 1);
21 }
22}

29void uart_set_baud_rate(UART_Handle_t* pUART_Handle){
30 uint32_t fclk = RCC_GetUSARTClock(pUART_Handle->pUARTx);
31 if(fclk == 0){
32 return;
33 }
34 uint32_t temp = fclk*125/(1*(2-pUART_Handle->UART_Config.UART_CONFIG_OVER_SAMPLING)*pUART_Handle->UART_Config.UART_CONFIG_BAUD_RATE);
35 uint32_t matissa = temp/1000;
36 uint32_t fraction= (temp-(matissa*1000))*8*(2-pUART_Handle->UART_Config.UART_CONFIG_OVER_SAMPLING)+500;
37 fraction = fraction / 1000;
38 pUART_Handle->pUARTx->USART_BRR &= ~(0xFFFF);
39 pUART_Handle->pUARTx->USART_BRR |= (matissa<<4);
40 pUART_Handle->pUARTx->USART_BRR |= (fraction);
41}

43void uart_init(UART_Handle_t* pUART_Handle)
44{
45 //0. Basic Check first
46 if((pUART_Handle == NULL) || (pUART_Handle->pUARTx == NULL)){
47 return;
48 }
49 //1.Enable the Peripheral Clock before writing to any of the Configuration Register
50 if(pUART_Handle->pUARTx == UART1){
51 USART1_PCLK_EN();
52 }else if(pUART_Handle->pUARTx == UART2){
53 USART2_PCLK_EN();
54 }else if(pUART_Handle->pUARTx == UART3){
55 USART3_PCLK_EN();
56 }else if(pUART_Handle->pUARTx == UART4){
57 UART4_PCLK_EN();
58 }else if(pUART_Handle->pUARTx == UART5){
59 UART5_PCLK_EN();
60 }else if(pUART_Handle->pUARTx == UART6){
61 USART6_PCLK_EN();
62 }
63
64 //2. Set Over Sampling (@note This step must come before setting the baud rate)
65 pUART_Handle->pUARTx->USART_CR1 &= ~(1<<USART_CR1_OVER8);
66 pUART_Handle->pUARTx->USART_CR1 |= (pUART_Handle->UART_Config.UART_CONFIG_OVER_SAMPLING<<USART_CR1_OVER8);
67
68 //3. Set the mode
69 if(pUART_Handle->UART_Config.UART_CONFIG_MODE == UART_CONFIG_MODE_RX){
70 pUART_Handle->pUARTx->USART_CR1 &= ~(1U << USART_CR1_TE);
71 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_RE);
72 }else if (pUART_Handle->UART_Config.UART_CONFIG_MODE == UART_CONFIG_MODE_TX){
73 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_TE);
74 pUART_Handle->pUARTx->USART_CR1 &= ~(1U << USART_CR1_RE);
75 }else if(pUART_Handle->UART_Config.UART_CONFIG_MODE == UART_CONFIG_MODE_TX_RX){
76 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_TE);
77 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_RE);
78 }
79
80 //4. Set Baud Rate
81 uart_set_baud_rate(pUART_Handle);
82
83 //5. Set the parity
84 if(pUART_Handle->UART_Config.UART_CONFIG_PARITY == UART_CONFIG_PARITY_NONE){
85 pUART_Handle->pUARTx->USART_CR1 &= ~(1U << USART_CR1_PCE);
86 }else if(pUART_Handle->UART_Config.UART_CONFIG_PARITY == UART_CONFIG_PARITY_EVEN ){
87 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_PCE);
88 pUART_Handle->pUARTx->USART_CR1 &= ~(1 << USART_CR1_PS);
89 }else if(pUART_Handle->UART_Config.UART_CONFIG_PARITY == UART_CONFIG_PARITY_ODD){
90 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_PCE);
91 pUART_Handle->pUARTx->USART_CR1 |= (1U << USART_CR1_PS);
92 }
93
94 //6. Set No of stop Bits
95 pUART_Handle->pUARTx->USART_CR2 &= ~(0x3 << USART_CR2_STOP);
96 pUART_Handle->pUARTx->USART_CR2 |= (pUART_Handle->UART_Config.UART_CONFIG_NO_OF_STOP_BITS << USART_CR2_STOP);
97
98 //7. Set Word Length
99 pUART_Handle->pUARTx->USART_CR1 &= ~(0x1<< USART_CR1_M);
100 pUART_Handle->pUARTx->USART_CR1 |= ((pUART_Handle->UART_Config.UART_CONFIG_WORD_LENGTH)<< USART_CR1_M);
101
102 //8. Set Hardware Flow
103 if(pUART_Handle->UART_Config.UART_CONFIG_HW_FLOW_CONTROL == UART_CONFIG_HW_FLOW_CONTROL_NONE){
104 pUART_Handle->pUARTx->USART_CR3 &= ~(1U<< USART_CR3_CTSE);
105 pUART_Handle->pUARTx->USART_CR3 &= ~(1U<< USART_CR3_RTSE);
106 }else if(pUART_Handle->UART_Config.UART_CONFIG_HW_FLOW_CONTROL == UART_CONFIG_HW_FLOW_CONTROL_CTS){
107 pUART_Handle->pUARTx->USART_CR3 |= (1U<< USART_CR3_CTSE);
108 pUART_Handle->pUARTx->USART_CR3 &= ~(1U<< USART_CR3_RTSE);
109 }else if(pUART_Handle->UART_Config.UART_CONFIG_HW_FLOW_CONTROL == UART_CONFIG_HW_FLOW_CONTROL_RTS){
110 pUART_Handle->pUARTx->USART_CR3 &= ~(1U<< USART_CR3_CTSE);
111 pUART_Handle->pUARTx->USART_CR3 |= (1U<< USART_CR3_RTSE);
112 }else if(pUART_Handle->UART_Config.UART_CONFIG_HW_FLOW_CONTROL == UART_CONFIG_HW_FLOW_CONTROL_CTS_RTS){
113 pUART_Handle->pUARTx->USART_CR3 |= (1U<< USART_CR3_CTSE);
114 pUART_Handle->pUARTx->USART_CR3 |= (1U<< USART_CR3_RTSE);
115 }

161void UART_SendData_Blocking(UART_Handle_t *pUARTHandle, uint8_t *TXBuffer, uint32_t Len) {
162 if(pUARTHandle == NULL || pUARTHandle->pUARTx == NULL || TXBuffer == NULL || Len == 0){
163 return;
164 }
165 uint32_t idx = 0;
166 while (Len > 0) {
167 if (UART_Get_Status_Flag(pUARTHandle->pUARTx, UART_STATUS_FLAG_TXE)) {
168 pUARTHandle->pUARTx->USART_SR &= ~(1U << USART_SR_TC);
169 uint8_t temp = TXBuffer[idx++];
170 pUARTHandle->pUARTx->USART_DR = temp;
171 Len--;
172 }
173 }
174 while (!UART_Get_Status_Flag(pUARTHandle->pUARTx, UART_STATUS_FLAG_TC)) {
175
176 }
177}

196uint32_t UART_Recieve_String(UART_Handle_t *pUARTHandle,char* str){
197 uint32_t count = 0;
198 uint8_t temp = 0;
199 while(temp != '\r'){
200 UART_RecieveData_Blocking(pUARTHandle, &temp, 1);
201 UART_Echo(pUARTHandle, (char)temp);
202 if(temp == '\b'){
203 count--;
204 continue;
205 }
206 str[count] = (char)temp;
207 count++;
208 }
209 return count-1;// Reduce the "\n"
210}

213 //First Identify the interrupt priority Number

215 if(pUARTHandle->pUARTx == UART1){
216 IRQNumber = IRQ_NUM_USART1;

226 IRQNumber = IRQ_NUM_USART6;
227 }else {
228 return;
229 }
230 uint32_t IPR_Number = IRQNumber / 4;
231 uint32_t IPR_Index = (IRQNumber % 4)*8;

235

238 NVIC->IPR[IPR_Number] |= (Priority << (IPR_Index+4));

274void UART_IRQHandling(UART_Handle_t *pUARTHandle){
275 // Extract out all the status bits and check which type of interrupt occurred using Status Reg and Control Reg
276 uint8_t CR1_TCIE_Bit = ((pUARTHandle->pUARTx->USART_CR1>>USART_CR1_TCIE) & (1 ));
277 uint8_t CR1_TXEIE_Bit = ((pUARTHandle->pUARTx->USART_CR1>>USART_CR1_TXEIE) & (1));
278 uint8_t CR1_PEIE_Bit = ((pUARTHandle->pUARTx->USART_CR1>>USART_CR1_PEIE) & (1));
279 uint8_t CR1_RXNEIE_Bit = ((pUARTHandle->pUARTx->USART_CR1>>USART_CR1_RXNEIE) & (1));
280 uint8_t CR1_IDLEIE_Bit = ((pUARTHandle->pUARTx->USART_CR1>>USART_CR1_IDLEIE) & (1));
281
282 uint8_t SR_TC_Bit = UART_Get_Status_Flag(pUARTHandle->pUARTx, USART_SR_TC);
283 uint8_t SR_TXE_Bit = UART_Get_Status_Flag(pUARTHandle->pUARTx, USART_SR_TXE);
284 uint8_t SR_PE_Bit = UART_Get_Status_Flag(pUARTHandle->pUARTx, USART_SR_TC);
285 uint8_t SR_RXNE_Bit = UART_Get_Status_Flag(pUARTHandle->pUARTx, USART_SR_RXNE);
286 uint8_t SR_IDLE_Bit = UART_Get_Status_Flag(pUARTHandle->pUARTx, USART_SR_IDLE);
287
288 // Based on SR and CR decide which interrupt occurred
289 if(CR1_TCIE_Bit & SR_TC_Bit){// TC interrupt Triggered
290 //@todo
291 }
292 if(CR1_TXEIE_Bit & SR_TXE_Bit){// TXE interrupt Triggered
293 // Load the TX data
294 if(pUARTHandle->TXLen > 0){
295 pUARTHandle->pUARTx->USART_DR = *(pUARTHandle->pTXBuffer);
296 pUARTHandle->pTXBuffer++;
297 pUARTHandle->TXLen--;
298 }
299 // If the length become zero close the communication
300 if(pUARTHandle->TXLen == 0){
301 //Disable TXNIE interrupt
302 UART_ITControl(pUARTHandle->pUARTx, UART_IT_TYPE_TXE, DISABLE);
303
304 //Clear the global Handle
305 pUARTHandle->TXState = UART_STATE_READY;
306 pUARTHandle->pTXBuffer = NULL;
307
308 //Notify the user app
309 UART_ApplicationEventCallback(pUARTHandle, UART_EVENT_TX_CMPLT);
310 }
311 }
312 if(CR1_PEIE_Bit & SR_PE_Bit){// PE interrupt Triggered
313 //@todo
314 }
315 if(CR1_RXNEIE_Bit & SR_RXNE_Bit){// RXNEIE interrupt Triggered
316 // Take out the data from the RX Reg, this will automatically clear the RXNEIE flag
317 uint8_t RxByte = pUARTHandle->pUARTx->USART_DR;
318 if(pUARTHandle->RXLen > 0){
319 *(pUARTHandle->pRXBuffer) = RxByte;
320 pUARTHandle->pRXBuffer++;
321 pUARTHandle->RXLen--;
322 }
323 // If RX length becomes zero it means it have received all the bytes and now we have to disable the interrupt i.e. we require no more interrupts
324 if(pUARTHandle->RXLen == 0){
325 // Disable the RXNE interrupt
326 UART_ITControl(pUARTHandle->pUARTx, UART_IT_TYPE_RXNE, DISABLE);
327
328 //Reset States
329 pUARTHandle->RXState = UART_STATE_READY;
330 pUARTHandle->pRXBuffer = NULL;
331
332 //Notify the user app
333 UART_ApplicationEventCallback(pUARTHandle, UART_EVENT_RX_CMPLT);
334 }
335 }
336 if(CR1_IDLEIE_Bit & SR_IDLE_Bit){// IDLE interrupt Triggered
337 //@todo
338 }
339}

341uint8_t UART_SendData_IT(UART_Handle_t* pUARTHandle, uint8_t* TXBuff, uint32_t Len){
342 /* Parameter validation */
343 if (pUARTHandle == NULL || TXBuff == NULL || Len == 0) {
344 return 1; /* Error */
345 }
346
347 /* Check if UART is already busy in TX */
348 if (pUARTHandle->TXState == UART_STATE_BUSY_TX) {
349 return 2; /* Busy */
350 }
351
352 // Save the data in Global Handle
353 pUARTHandle->TXLen = Len;
354 pUARTHandle->pTXBuffer = TXBuff;
355 pUARTHandle->TXState = UART_STATE_BUSY_TX;
356
357 //Enable TXE interrupt
358 UART_ITControl(pUARTHandle->pUARTx, UART_IT_TYPE_TXE, ENABLE);
359
360 // On success return 0;
361 return 0;
362}

364uint8_t UART_ReceiveData_IT(UART_Handle_t* pUARTHandle, uint8_t* RXBuff, uint32_t Len){
365 /* Parameter validation */
366 if (pUARTHandle == NULL || RXBuff == NULL || Len == 0) {
367 return 1; /* Error */
368 }
369
370 /* Check if UART is already busy in RX */
371 if (pUARTHandle->RXState == UART_STATE_BUSY_RX) {
372 return 2; /* Busy */
373 }
374
375 // Save the data in Global Handle
376 pUARTHandle->RXLen = Len;
377 pUARTHandle->RXState = UART_STATE_BUSY_RX;
378 pUARTHandle->pRXBuffer = RXBuff;
379
380 /* Enable RXNE interrupt at UART level */
381 UART_ITControl(pUARTHandle->pUARTx, UART_IT_TYPE_RXNE, ENABLE);
382
383 // On success Return 0
384 return 0;
385}

387void UART_ITControl(UART_Reg_Def_t *pUARTx, uint8_t InterruptType, uint8_t EnOrDi) {
388 if (pUARTx == NULL) {
389 return;
390 }
391
392 if (EnOrDi == ENABLE) {
393 switch (InterruptType) {
394 case UART_IT_TYPE_TXE:
395 pUARTx->USART_CR1 |= (1 << USART_CR1_TXEIE);
396 break;
397
398 case UART_IT_TYPE_TC:
399 pUARTx->USART_CR1 |= (1 << USART_CR1_TCIE);
400 break;
401
402 case UART_IT_TYPE_RXNE:
403 pUARTx->USART_CR1 |= (1 << USART_CR1_RXNEIE);
404 break;
405
406 default:
407 break;
408 }
409 } else {
410 switch (InterruptType) {
411 case UART_IT_TYPE_TXE:
412 pUARTx->USART_CR1 &= ~(1 << USART_CR1_TXEIE);
413 break;
414
415 case UART_IT_TYPE_TC:
416 pUARTx->USART_CR1 &= ~(1 << USART_CR1_TCIE);
417 break;
418
419 case UART_IT_TYPE_RXNE:
420 pUARTx->USART_CR1 &= ~(1 << USART_CR1_RXNEIE);
421 break;
422
423 default:
424 break;
425 }
426 }
