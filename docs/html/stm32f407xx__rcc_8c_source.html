18static uint32_t RCC_Get_PLL_CLK(void){
19 //1.Identify the PLL Source using PPLSRC bit of PLLCFGR register
20 uint8_t PLL_SRC = (RCC->PLLCFGR >> 22) & 0x1 ;
21 uint32_t PLL_CLK_Freq = 0;
22 if(PLL_SRC == 0){ // PLL Src is HSI
23 PLL_CLK_Freq = HSI_Freq;
24 }else if(PLL_SRC == 1){// PLL Src is HSE
25 PLL_CLK_Freq = HSE_Freq;
26 }
27 //2. Extract PLLM, PLLN, PLLP factors
28 uint8_t PLLM = RCC->PLLCFGR & 0x3f;
29 if(PLLM < 2){
30 return 0;
31 }
32 uint16_t PLLN = (RCC->PLLCFGR >> 6) & 0x1FF;
33 if(!(PLLN>=50 && PLLN <=432)){
34 return 0;
35 }
36 uint8_t PLLP = (RCC->PLLCFGR>>16) & 0x3;
37 if(PLLP==0){
38 PLLP = 2;
39 }else if(PLLP==1){
40 PLLP = 4;
41 }else if(PLLP==2){
42 PLLP = 6;
43 }else if(PLLP==3){
44 PLLP = 8;
45 }
46
47 return (PLL_CLK_Freq / PLLM) * PLLN / PLLP;
48}

50static uint32_t RCC_Get_SYSCLK_Freq(void){
51 //1. First Identify the SYS CLOCK Source
52 uint8_t clk_src = (RCC->CFGR>>2) & 0x3;
53
54 //2. Get the AHB Prescaler
55 uint32_t AHB_Prescaler = AHB_PRESCALER_TABLE[(RCC->CFGR>>4) & 0xF];
56 if(clk_src == CLK_SRC_HSI){//SRC is HSI
57 return HSI_Freq/AHB_Prescaler;
58 }else if(clk_src == CLK_SRC_HSE){// SRC is HSE
59 return HSE_Freq/AHB_Prescaler;
60 }else if(clk_src == CLK_SRC_PLL){//SRC is PLL
61 uint32_t PLL_CLK = RCC_Get_PLL_CLK();
62 if(PLL_CLK == 0){
63 return 0;
64 }
65 return (PLL_CLK/AHB_Prescaler);
66 }
67 return 0;
68}
