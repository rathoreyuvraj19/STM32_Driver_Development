16static void SPI_Tx_IRQ_Helper(SPIx_Handle_t *pHandle){
17 // Check and Load the data into DR
18 if (pHandle->TXLen > 0) {
19 pHandle->pSPIx->DR = *(pHandle->pTXBuffer);
20 pHandle->TXLen--;
21 pHandle->pTXBuffer++;
22 }
23
24 // Check for transmission complete
25 if (pHandle->TXLen == 0) {
26 SPIx_PeripheralInterruptControl(pHandle->pSPIx, SPI_IT_TYPE_TXE, DISABLE);
27 pHandle->TxState = SPI_STATE_READY;
28 pHandle->TXLen = 0;
29 pHandle->pTXBuffer = NULL;
30 SPI_ApplicationEventCallback(pHandle, SPI_EVENT_TX_CMPLT);
31 }
32}

59 //2. Inform the application about the over run error
60 SPI_ApplicationEventCallback(pHandle, SPI_EVENT_OVR_ERR);

72void SPIx_Init(SPIx_Handle_t *pSPI_Handle){
73 SPIx_RegDef_t* pSPIx = pSPI_Handle->pSPIx;
74 uint8_t SPI_DEVICE_MODE = pSPI_Handle->SPI_CONFIG.SPI_DEVICE_MODE;
75 uint8_t SPI_BUS_MODE = pSPI_Handle->SPI_CONFIG.SPI_BUS_MODE;
76 uint8_t SPI_CLOCK_SPEED = pSPI_Handle->SPI_CONFIG.SPI_CLOCK_SPEED;
77 uint8_t SPI_CPOL = pSPI_Handle->SPI_CONFIG.SPI_CPOL;
78 uint8_t SPI_CPHA = pSPI_Handle->SPI_CONFIG.SPI_CPHA;
79 uint8_t SPI_FRAME_SIZE = pSPI_Handle->SPI_CONFIG.SPI_FRAME_SIZE;
80 uint8_t SPI_SSM_SETTING = pSPI_Handle->SPI_CONFIG.SPI_SSM_SETTING;
81 uint8_t SPI_BIT_ORDER = pSPI_Handle->SPI_CONFIG.SPI_BIT_ORDER;
82 uint8_t SPI_SSOE = pSPI_Handle->SPI_CONFIG.SPI_SSOE;
83 uint8_t SPI_CRC_EN = pSPI_Handle->SPI_CONFIG.SPI_CRC_EN;
84 //@todo use SPI_CRC_EN later
85 (void)SPI_CRC_EN;
87 RCC_RegDef_t* pRCC = RCC;
88 if(pSPIx == SPI1){
89 pRCC->APB2ENR |= (1 << 12);
90 }else if(pSPIx == SPI2){
91 pRCC->APB1ENR |= (1 << 14);
92 }else if(pSPIx == SPI3){
93 pRCC->APB1ENR |= (1 << 15);
94 }
95
97 pSPIx->CR1 &= ~(1 << SPI_CR1_MSTR_Pos);
98 pSPIx->CR1 |= (SPI_DEVICE_MODE << SPI_CR1_MSTR_Pos);
99
101 if (SPI_BUS_MODE == SPI_BUS_MODE_FULL_DUPLEX) { // Normal Mode
102 //Clear BIDIMODE Bit in CR1 Reg
103 pSPIx->CR1 &= ~(1 << SPI_CR1_BIDIMODE_Pos);
104 // Also Clear the RXONLY Bit just for safety
105 pSPIx->CR1 &= ~(1 << SPI_CR1_RXONLY_Pos);
106 }else if(SPI_BUS_MODE == SPI_BUS_MODE_HALF_DUPLEX){
107 //Set BIDIMODE Bit in CR1 Reg
109 pSPIx->CR1 |= (1 << SPI_CR1_BIDIMODE_Pos);
110 }else if(SPI_BUS_MODE == SPI_BUS_MODE_SIMPLEX_RX){// Simplex Means either Tx only or Rx Only
115 pSPIx->CR1 &= ~(1 << SPI_CR1_BIDIMODE_Pos);
116 pSPIx->CR1 |= (1 << SPI_CR1_RXONLY_Pos);
117 }
118
120 pSPIx->CR1 &= ~(0x07 << 3);
121 pSPIx->CR1 |= (SPI_CLOCK_SPEED << 3);
122
127 pSPIx->CR1 &= ~(1 << SPI_CR1_CPHA_Pos);
128 pSPIx->CR1 &= ~(1 << SPI_CR1_CPOL_Pos);
129 pSPIx->CR1 |= (SPI_CPHA << SPI_CR1_CPHA_Pos);
130 pSPIx->CR1 |= (SPI_CPOL << SPI_CR1_CPOL_Pos);
131
133 pSPIx->CR1 &= ~(1 << SPI_CR1_DFF_Pos);
134 pSPIx->CR1 |= (SPI_FRAME_SIZE << SPI_CR1_DFF_Pos);
135
137 pSPIx->CR1 &= ~(1 << SPI_CR1_LSBFIRST_Pos);
138 pSPIx->CR1 |= (SPI_BIT_ORDER << SPI_CR1_LSBFIRST_Pos);
139
146 pSPIx->CR1 &= ~(1 << SPI_CR1_SSM_Pos); // Clear the SSM Bit in CR1 Reg
147 pSPIx->CR1 |= (SPI_SSM_SETTING << SPI_CR1_SSM_Pos);// Set the bit
148
149 pSPIx->CR2 &= ~(1 << SPI_CR2_SSOE_Pos); // Clear the SSOE Bit in CR2 Reg
150 pSPIx->CR2 |= (SPI_SSOE << SPI_CR2_SSOE_Pos);// Set the bit

191void SPIx_Peri_Control(SPIx_RegDef_t *pSPIx, uint8_t EN_DI){
192 if(EN_DI == ENABLE){
193 pSPIx->CR1 |= (1 << SPI_CR1_SPE_Pos);
194 }else{
195 // 1. Wait until TXE=1
196 while(!SPIx_GetFlagStatus(pSPIx, SPI_STATUS_FLAG_TXE));
197 // 2. Then wait until BSY=0
198 while(SPIx_GetFlagStatus(pSPIx, SPI_STATUS_FLAG_BSY));
199 // 3. The Disable the SPI
200 pSPIx->CR1 &= ~(1 << SPI_CR1_SPE_Pos);
201 }
202}

208void SPIx_SendData_Blocking(SPIx_RegDef_t *pSPIx, uint8_t* pTX_Data_Buffer, uint32_t Len_Byte){
210 while (Len_Byte > 0) {
211 // Wait until TXE = 1
212 while (!SPIx_GetFlagStatus(pSPIx, SPI_STATUS_FLAG_TXE));
213 //Check the data format
214 if (pSPIx->CR1 & (1U << SPI_CR1_DFF_Pos)) { // Frame Size = 16
215 pSPIx->DR = *((uint16_t*) pTX_Data_Buffer);
216 pTX_Data_Buffer += 2;
217 Len_Byte -= 2;
218 } else { // Frame Size = 8
219 pSPIx->DR = *pTX_Data_Buffer;
220 pTX_Data_Buffer++;
221 Len_Byte--;
222 }
223 }
224
225
231 while (!SPIx_GetFlagStatus(pSPIx, SPI_STATUS_FLAG_TXE));
232 while ( SPIx_GetFlagStatus(pSPIx, SPI_STATUS_FLAG_BSY));
233}

235void SPIx_ReceiveData_Blocking(SPIx_RegDef_t *pSPIx, uint8_t *pRX_Data_Buffer, uint32_t Len_Byte)
236{
238 while (Len_Byte > 0)
239 {
240 if(pSPIx->CR1 & (1U << SPI_CR1_DFF_Pos)){
242 while (!SPIx_GetFlagStatus(pSPIx, SPI_STATUS_FLAG_TXE));
243
245 pSPIx->DR = (uint16_t)0xFFFF;
246
248 while (!SPIx_GetFlagStatus(pSPIx, SPI_STATUS_FLAG_RXNE));
249
251 *(uint16_t*)pRX_Data_Buffer = (uint16_t)pSPIx->DR;
252 pRX_Data_Buffer+=2;
253 Len_Byte-=2;
254 }else{
256 while (!SPIx_GetFlagStatus(pSPIx, SPI_STATUS_FLAG_TXE));
257
259 pSPIx->DR = (uint8_t)0xFF;
260
262 while (!SPIx_GetFlagStatus(pSPIx, SPI_STATUS_FLAG_RXNE));
263
265 *pRX_Data_Buffer = (uint8_t)pSPIx->DR;
266 pRX_Data_Buffer++;
267 Len_Byte--;
268 }
269 }
270
272 while (!SPIx_GetFlagStatus(pSPIx, SPI_STATUS_FLAG_TXE));
273 while (SPIx_GetFlagStatus(pSPIx, SPI_STATUS_FLAG_BSY));
274}

276void SPIx_IRQConfig(uint8_t IRQNumber, uint8_t irq_priority){
277 uint8_t IPR_Reg_Number = IRQNumber / 4;
278 uint8_t IRQ_Pos = (IRQNumber % 4) * 8;
279 NVIC_RegDef_t *pNVIC = (NVIC_RegDef_t *)NVIC_BASEADDR;
280 irq_priority &= 0x0F; // Masking the priority
281 if(IRQNumber == IRQ_NUM_SPI1 || IRQNumber == IRQ_NUM_SPI2 || IRQNumber == IRQ_NUM_SPI3 ){
282 pNVIC->IPR[IPR_Reg_Number] &= ~(0xFF << IRQ_Pos);
283 pNVIC->IPR[IPR_Reg_Number] |= (((uint32_t)irq_priority) << (IRQ_Pos + 4)); // We Just have to write upper 4 bits
284 }else {
285 return ;
286 }
287}

289void SPIx_IRQControl(uint8_t IRQNumber, uint8_t EN_DI){
290 //1. Check if IRQ is not spi related, if not return
291 if (IRQNumber != IRQ_NUM_SPI1 &&
292 IRQNumber != IRQ_NUM_SPI2 &&
293 IRQNumber != IRQ_NUM_SPI3) {
294 return;
295 }
296
297 uint32_t reg_Idx = IRQNumber / 32;
298 uint32_t bit_position = IRQNumber % 32;
299 if(EN_DI == ENABLE){
300 // Enable interrupt in NVIC using ISER
301 NVIC->ISER[reg_Idx] |= (1U << bit_position);
302 }else{
303 // To disable the interrupt write 1 to ICER register
304 NVIC->ICER[reg_Idx] |= (1U << bit_position);
305 // Also clear the pending interrupt using ICPR
306 NVIC->ICPR[reg_Idx] |= (1U << bit_position);
307 }
308}

310void SPIx_PeripheralInterruptControl(SPIx_RegDef_t *pSPIx, uint8_t ITType, uint8_t EN_DI){
311 if (EN_DI) {
312 if (ITType == SPI_IT_TYPE_TXE) {
313 pSPIx->CR2 |= (1U << SPI_CR2_TXEIE_Pos);
314 } else if (ITType == SPI_IT_TYPE_RXNE) {
315 pSPIx->CR2 |= (1U << SPI_CR2_RXNEIE_Pos);
316 } else if (ITType == SPI_IT_TYPE_ERR) {
317 pSPIx->CR2 |= (1U << SPI_CR2_ERRIE_Pos);
318 }
319 } else {
320 if (ITType == SPI_IT_TYPE_TXE) {
321 pSPIx->CR2 &= ~(1U << SPI_CR2_TXEIE_Pos);
322 } else if (ITType == SPI_IT_TYPE_RXNE) {
323 pSPIx->CR2 &= ~(1U << SPI_CR2_RXNEIE_Pos);
324 } else if (ITType == SPI_IT_TYPE_ERR) {
325 pSPIx->CR2 &= ~(1U << SPI_CR2_ERRIE_Pos);

330uint8_t SPIx_SendDataIT(SPIx_Handle_t *pHandle, uint8_t *pTxBuf, uint32_t Len){
332 //0. Basic Checks first
333 if(pHandle == NULL || pTxBuf == NULL || Len == 0 ){
334 return SPI_INVALID_PARAM;
335 }
336 if(pHandle->TxState == SPI_STATE_BUSY_IN_TX || pHandle->RxState == SPI_STATE_BUSY_IN_RX){
337 return SPI_BUSY;
338 }
339 //1. Save TXBuffer and Length Info in some global variable
340 pHandle->pTXBuffer = pTxBuf;
341 pHandle->TXLen = Len;
342
343 //2. Mark the SPI as Busy in transmission so that no other code can
344 // take over the SPI peripheral until transmission is over
345 pHandle->TxState = SPI_STATE_BUSY_IN_TX;
346 //3. Enable the TXEIE control bit in the CR2 reg to generate interrupt whenever the TX buffer of the corresponding
347 // SPI is empty (i.e. Whenever the TXE flag is set in the status register)
348 SPIx_PeripheralInterruptControl(pHandle->pSPIx, SPI_IT_TYPE_TXE, ENABLE);
349 return SPI_TXRX_SUCCESS;
350 //4. @todo Data Transmission will be handled by the ISR code
351}

353uint8_t SPIx_ReceiveDataIT(SPIx_Handle_t *pHandle, uint8_t *pRxBuf, uint32_t Len){
355 //0. Basic Check first
356 if(pHandle->TxState == SPI_STATE_BUSY_IN_TX || pHandle->RxState == SPI_STATE_BUSY_IN_RX){
357 return SPI_BUSY;
358 }
359 if(pHandle == NULL || pRxBuf == NULL || Len ==0){
360 return SPI_INVALID_PARAM;
361 }
362 //1. Save the RxBuffer and Length info in some global variable
363 pHandle->RXLen = Len;
364 pHandle->pRXBuffer = pRxBuf;
365
366 //2. Mark the SPI as Busy in Rx so that no other code can
367 // take over the SPI peripheral until the rx is over
368 pHandle->RxState = SPI_STATE_BUSY_IN_RX;
369 //3. Enable the RXNEIE control bit in the CR2 reg to generate interrupt whenever the RX buffer of the
370 //corresponding SPI is not empty (i.e. Whenever the RXNE flag is set in the status register)
371 SPIx_PeripheralInterruptControl(pHandle->pSPIx, SPI_IT_TYPE_RXNE, ENABLE);
372 return SPI_TXRX_SUCCESS;
373 //4. @todo Data Reception will be handled by the ISR code
374}

376void SPIx_IRQHandling(SPIx_Handle_t *pHandle){
377
378 /* basic sanity check */
379 if (pHandle == NULL || pHandle->pSPIx == NULL) {
380 return;
381 }
382
383
384 //0. Extract the status status flags
385 //uint8_t txe_flag = (pHandle->pSPIx->SR & (1 << SPI_SR_TXE_Pos));
386 uint8_t txe_flag = SPIx_GetFlagStatus(pHandle->pSPIx, SPI_STATUS_FLAG_TXE);
387 //uint8_t rxne_flag = (pHandle->pSPIx->SR & (1 << SPI_SR_RXNE_Pos));
388 uint8_t rxne_flag = SPIx_GetFlagStatus(pHandle->pSPIx, SPI_STATUS_FLAG_RXNE);
389 //uint8_t ovr_flag = (pHandle->pSPIx->SR & (1 << SPI_SR_OVR_Pos));
390 uint8_t ovr_flag = SPIx_GetFlagStatus(pHandle->pSPIx, SPI_STATUS_FLAG_OVR);
391 uint8_t crc_err_flag = SPIx_GetFlagStatus(pHandle->pSPIx, SPI_STATUS_FLAG_CRCERR);
392 uint8_t modf_flag = SPIx_GetFlagStatus(pHandle->pSPIx, SPI_STATUS_FLAG_MODF);
393
394 uint8_t txeie_bit = (pHandle->pSPIx->CR2 & (1 << SPI_CR2_TXEIE_Pos));
395 uint8_t rxneie_bit = (pHandle->pSPIx->CR2 & (1 << SPI_CR2_RXNEIE_Pos));
396 uint8_t errie_bit = (pHandle->pSPIx->CR2 & (1 << SPI_CR2_ERRIE_Pos));
397
398
399 // Check TXE flag
400 if(txe_flag && txeie_bit){
401 SPI_Tx_IRQ_Helper(pHandle);
402 }
403
404 // Check RXNE flag
405 if(rxne_flag && rxneie_bit){
406 SPI_Rx_IRQ_Helper(pHandle);
407 }
408
409 // Check Overrun Error
410 if (ovr_flag && errie_bit)
411 {
412 SPI_OVR_Err_IRQ_Helper(pHandle);
413 }
414
415 // Check CRC Error
416 if (crc_err_flag && errie_bit)
417 {
418 SPI_CRC_Err_IRQ_Helper(pHandle);
419 }
420
421 // Check Master mode fault Error
422 if (modf_flag && errie_bit)
423 {
424 SPI_MODF_Err_IRQ_Helper(pHandle);
425 }
426}
